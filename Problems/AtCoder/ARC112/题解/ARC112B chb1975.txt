#include <bits/stdc++.h>
#define rep(i,n) for ((i)=1;(i)<=(n);(i)++)
using namespace std;
long long b,c,ans;
int main(){
	cin>>b>>c;
	vector<pair<long long,long long> > v;
	v.push_back(make_pair(b-c/2,b));
	if(c>=1) v.push_back(make_pair(-b-(c-1)/2,-b));
	if(c>=1) v.push_back(make_pair(-b,-(b-(c-1)/2)));
	if(c>=2) v.push_back(make_pair(b,b+(c-2)/2));
	sort(v.begin(),v.end());
	long long lst=-0x3f3f3f3f3f3f3f3fll;
	int i;
	for(i=0;i<v.size();i++){
		lst=max(lst,v[i].first);
		if(v[i].second>=lst){
			ans+=v[i].second-lst+1;
			lst=v[i].second+1;
		}
	}
	cout<<ans<<endl;
	return 0;
}
